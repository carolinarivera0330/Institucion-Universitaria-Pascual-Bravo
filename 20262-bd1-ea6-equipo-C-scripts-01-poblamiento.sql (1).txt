-- 1. Usuarios (100 registros)
INSERT INTO usuario (id_rol, id_tipo_usuario, nombre_usuario, correo_electronico, contrasena_hash, fecha_registro, telefono_contacto)
SELECT 
    CASE 
        WHEN g <= 2 THEN 1
        WHEN g <= 5 THEN 2
        WHEN g <= 90 THEN 3
        ELSE 4
    END AS id_rol,
    ((g - 1) % 9) + 1 AS id_tipo_usuario,
    (ARRAY['camila','juan','mateo','valentina','santiago','sofia','daniel','laura','andres','paula','carlos','natalia','felipe','mariana','david','juliana'])[1 + floor(random()*16)::int] 
        || '_' || 
    (ARRAY['gomez','rodriguez','lopez','martinez','garcia','perez','sanchez','ramirez','torres','flores','diaz','vargas','castro','ruiz','ortiz','morales'])[1 + floor(random()*16)::int] 
        || '_' || g AS nombre_usuario,
    'usuario' || g || '@pascualbravo.edu.co' AS correo_electronico,
    md5('Pascual2026*' || g) AS contrasena_hash,
    CURRENT_TIMESTAMP - (random() * interval '365 days') AS fecha_registro,
    '3' || lpad(floor(random()*999999999)::text, 9, '0') AS telefono_contacto
FROM generate_series(1, 100) AS g;

-- 15. Transacciones y Detalle de Transacción (100 registros protegidos contra NULL)
DO $$
DECLARE
    v_id_transaccion BIGINT;
    v_id_comprador BIGINT;
    v_id_vendedor BIGINT;
    v_id_producto BIGINT;
    v_id_servicio BIGINT;
    v_precio NUMERIC;
    v_cantidad INT;
    i INT;
BEGIN
    FOR i IN 1..100 LOOP
        v_cantidad := 1 + floor(random()*3)::int;


        IF (i % 2 = 0) THEN
            -- Seleccionar primero un producto real
            SELECT id_producto, id_usuario_vendedor, precio_producto 
            INTO v_id_producto, v_id_vendedor, v_precio 
            FROM producto ORDER BY random() LIMIT 1;


            -- Seleccionar un comprador distinto al vendedor
            SELECT id_usuario INTO v_id_comprador 
            FROM usuario WHERE id_rol = 3 AND id_usuario <> v_id_vendedor ORDER BY random() LIMIT 1;
            
            INSERT INTO transaccion (id_usuario_comprador, id_medio_pago, monto_total, fecha_transaccion)
            VALUES (v_id_comprador, (SELECT id_medio_pago FROM medio_pago ORDER BY random() LIMIT 1), COALESCE(v_precio, 10000) * v_cantidad, CURRENT_TIMESTAMP - (random() * interval '60 days'))
            RETURNING id_transaccion INTO v_id_transaccion;


            INSERT INTO detalle_transaccion (id_transaccion, id_producto, id_servicio, cantidad, precio_unitario)
            VALUES (v_id_transaccion, v_id_producto, NULL, v_cantidad, COALESCE(v_precio, 10000));
        ELSE
            -- Seleccionar primero un servicio real
            SELECT id_servicio, id_usuario_oferente, precio_servicio 
            INTO v_id_servicio, v_id_vendedor, v_precio 
            FROM servicio ORDER BY random() LIMIT 1;


            -- Seleccionar un comprador distinto al oferente
            SELECT id_usuario INTO v_id_comprador 
            FROM usuario WHERE id_rol = 3 AND id_usuario <> v_id_vendedor ORDER BY random() LIMIT 1;
            
            INSERT INTO transaccion (id_usuario_comprador, id_medio_pago, monto_total, fecha_transaccion)
            VALUES (v_id_comprador, (SELECT id_medio_pago FROM medio_pago ORDER BY random() LIMIT 1), COALESCE(v_precio, 15000) * v_cantidad, CURRENT_TIMESTAMP - (random() * interval '60 days'))
            RETURNING id_transaccion INTO v_id_transaccion;


            INSERT INTO detalle_transaccion (id_transaccion, id_producto, id_servicio, cantidad, precio_unitario)
            VALUES (v_id_transaccion, NULL, v_id_servicio, v_cantidad, COALESCE(v_precio, 15000));
        END IF;
    END LOOP;
END $$;
