--
-- Scripts de Creación de la Base de Datos
--
-- Todas las instrucciones se DEBEN EJECUTAR EN SECUENCIA SIN ERRORES
-- NOTA: Ojo con las tablas relacionadas. Primero las independientes y después las dependientes
--

--
-- Creación de las tablas
-- 

-- ----------------------------------------------------------------------------
-- 1. TABLAS INDEPENDIENTES (CATÁLOGOS BASE)
-- ----------------------------------------------------------------------------

CREATE TABLE rol (
    id_rol SERIAL PRIMARY KEY,
    nombre_rol VARCHAR(50) NOT NULL UNIQUE,
    descripcion_rol VARCHAR(255)
);

CREATE TABLE tipo_usuario (
    id_tipo_usuario SERIAL PRIMARY KEY,
    nombre_tipo_usuario VARCHAR(50) NOT NULL UNIQUE,
    descripcion_tipo_usuario VARCHAR(255)
);

CREATE TABLE categoria_publicacion (
    id_categoria_publicacion SERIAL PRIMARY KEY,
    nombre_categoria VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE tipo_reaccion (
    id_tipo_reaccion SERIAL PRIMARY KEY,
    nombre_reaccion VARCHAR(30) NOT NULL UNIQUE
);

CREATE TABLE tipo_grupo (
    id_tipo_grupo SERIAL PRIMARY KEY,
    nombre_tipo_grupo VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE tipo_servicio (
    id_tipo_servicio SERIAL PRIMARY KEY,
    nombre_tipo_servicio VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE tipo_producto (
    id_tipo_producto SERIAL PRIMARY KEY,
    nombre_tipo_producto VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE medio_pago (
    id_medio_pago SERIAL PRIMARY KEY,
    nombre_medio_pago VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE tipo_evento (
    id_tipo_evento SERIAL PRIMARY KEY,
    nombre_tipo_evento VARCHAR(50) NOT NULL UNIQUE
);


-- ----------------------------------------------------------------------------
-- 2. USUARIOS Y PERFILES
-- ----------------------------------------------------------------------------

CREATE TABLE usuario (
    id_usuario BIGSERIAL PRIMARY KEY,
    id_rol INTEGER NOT NULL,
    id_tipo_usuario INTEGER NOT NULL,
    nombre_usuario VARCHAR(50) NOT NULL UNIQUE,
    correo_electronico VARCHAR(100) NOT NULL UNIQUE,
    contrasena_hash VARCHAR(255) NOT NULL,
    fecha_registro TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_usuario_rol FOREIGN KEY (id_rol) REFERENCES rol(id_rol) ON DELETE RESTRICT,
    CONSTRAINT fk_usuario_tipo FOREIGN KEY (id_tipo_usuario) REFERENCES tipo_usuario(id_tipo_usuario) ON DELETE RESTRICT
);

CREATE TABLE perfil (
    id_perfil BIGSERIAL PRIMARY KEY,
    id_usuario BIGINT NOT NULL UNIQUE,
    datos_perfil JSONB,
    fecha_actualizacion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_perfil_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE
);


-- ----------------------------------------------------------------------------
-- 3. PUBLICACIONES, COMENTARIOS, REACCIONES Y MULTIMEDIA
-- ----------------------------------------------------------------------------

CREATE TABLE publicacion (
    id_publicacion BIGSERIAL PRIMARY KEY,
    id_usuario BIGINT NOT NULL,
    id_categoria_publicacion INTEGER NOT NULL,
    contenido_texto TEXT NOT NULL,
    fecha_publicacion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_publicacion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_publicacion_categoria FOREIGN KEY (id_categoria_publicacion) REFERENCES categoria_publicacion(id_categoria_publicacion) ON DELETE RESTRICT
);

CREATE TABLE comentario (
    id_comentario BIGSERIAL PRIMARY KEY,
    id_publicacion BIGINT NOT NULL,
    id_usuario BIGINT NOT NULL,
    contenido_texto TEXT NOT NULL,
    fecha_comentario TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_comentario_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion) ON DELETE CASCADE,
    CONSTRAINT fk_comentario_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE
);

CREATE TABLE reaccion (
    id_publicacion BIGINT NOT NULL,
    id_usuario BIGINT NOT NULL,
    id_tipo_reaccion INTEGER NOT NULL,
    fecha_reaccion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_publicacion, id_usuario),
    CONSTRAINT fk_reaccion_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion) ON DELETE CASCADE,
    CONSTRAINT fk_reaccion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_reaccion_tipo FOREIGN KEY (id_tipo_reaccion) REFERENCES tipo_reaccion(id_tipo_reaccion) ON DELETE RESTRICT
);

CREATE TABLE multimedia (
    id_multimedia BIGSERIAL PRIMARY KEY,
    id_publicacion BIGINT NOT NULL,
    tipo_archivo VARCHAR(30) NOT NULL,
    url_archivo VARCHAR(2048) NOT NULL,
    CONSTRAINT fk_multimedia_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion) ON DELETE CASCADE
);


-- ----------------------------------------------------------------------------
-- 4. GRUPOS Y RELACIÓN PIVOTE M:N
-- ----------------------------------------------------------------------------

CREATE TABLE grupo (
    id_grupo BIGSERIAL PRIMARY KEY,
    id_usuario_creador BIGINT NOT NULL,
    id_tipo_grupo INTEGER NOT NULL,
    nombre_grupo VARCHAR(100) NOT NULL,
    descripcion_grupo TEXT,
    fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_grupo_creador FOREIGN KEY (id_usuario_creador) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_grupo_tipo FOREIGN KEY (id_tipo_grupo) REFERENCES tipo_grupo(id_tipo_grupo) ON DELETE RESTRICT
);

CREATE TABLE usuario_grupo (
    id_usuario BIGINT NOT NULL,
    id_grupo BIGINT NOT NULL,
    fecha_ingreso TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_grupo),
    CONSTRAINT fk_ug_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_ug_grupo FOREIGN KEY (id_grupo) REFERENCES grupo(id_grupo) ON DELETE CASCADE
);


-- ----------------------------------------------------------------------------
-- 5. MARKETPLACE: SERVICIOS Y PRODUCTOS
-- ----------------------------------------------------------------------------

CREATE TABLE servicio (
    id_servicio BIGSERIAL PRIMARY KEY,
    id_usuario_oferente BIGINT NOT NULL,
    id_tipo_servicio INTEGER NOT NULL,
    descripcion_servicio TEXT NOT NULL,
    precio_servicio NUMERIC(12,2) NOT NULL CHECK (precio_servicio >= 0),
    CONSTRAINT fk_servicio_oferente FOREIGN KEY (id_usuario_oferente) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_servicio_tipo FOREIGN KEY (id_tipo_servicio) REFERENCES tipo_servicio(id_tipo_servicio) ON DELETE RESTRICT
);

CREATE TABLE producto (
    id_producto BIGSERIAL PRIMARY KEY,
    id_usuario_vendedor BIGINT NOT NULL,
    id_tipo_producto INTEGER NOT NULL,
    nombre_producto VARCHAR(100) NOT NULL,
    precio_producto NUMERIC(12,2) NOT NULL CHECK (precio_producto >= 0),
    CONSTRAINT fk_producto_vendedor FOREIGN KEY (id_usuario_vendedor) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_producto_tipo FOREIGN KEY (id_tipo_producto) REFERENCES tipo_producto(id_tipo_producto) ON DELETE RESTRICT
);


-- ----------------------------------------------------------------------------
-- 6. TRANSACCIONES Y TRAZA DETALLADA
-- ----------------------------------------------------------------------------

CREATE TABLE transaccion (
    id_transaccion BIGSERIAL PRIMARY KEY,
    id_usuario_comprador BIGINT NOT NULL,
    id_medio_pago INTEGER NOT NULL,
    monto_total NUMERIC(14,2) NOT NULL CHECK (monto_total >= 0),
    fecha_transaccion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_transaccion_comprador FOREIGN KEY (id_usuario_comprador) REFERENCES usuario(id_usuario) ON DELETE RESTRICT,
    CONSTRAINT fk_transaccion_medio FOREIGN KEY (id_medio_pago) REFERENCES medio_pago(id_medio_pago) ON DELETE RESTRICT
);

CREATE TABLE detalle_transaccion (
    id_detalle BIGSERIAL PRIMARY KEY,
    id_transaccion BIGINT NOT NULL,
    id_producto BIGINT,
    id_servicio BIGINT,
    cantidad INTEGER NOT NULL DEFAULT 1 CHECK (cantidad > 0),
    precio_unitario NUMERIC(12,2) NOT NULL CHECK (precio_unitario >= 0),
    CONSTRAINT fk_detalle_transaccion FOREIGN KEY (id_transaccion) REFERENCES transaccion(id_transaccion) ON DELETE CASCADE,
    CONSTRAINT fk_detalle_producto FOREIGN KEY (id_producto) REFERENCES producto(id_producto) ON DELETE SET NULL,
    CONSTRAINT fk_detalle_servicio FOREIGN KEY (id_servicio) REFERENCES servicio(id_servicio) ON DELETE SET NULL
);


-- ----------------------------------------------------------------------------
-- 7. EVENTOS, ASISTENCIAS, MENTORÍAS Y NOTIFICACIONES
-- ----------------------------------------------------------------------------

CREATE TABLE evento (
    id_evento BIGSERIAL PRIMARY KEY,
    id_usuario_organizador BIGINT NOT NULL,
    id_tipo_evento INTEGER NOT NULL,
    nombre_evento VARCHAR(100) NOT NULL,
    descripcion_evento TEXT,
    fecha_evento TIMESTAMPTZ NOT NULL,
    ubicacion_evento VARCHAR(255),
    CONSTRAINT fk_evento_organizador FOREIGN KEY (id_usuario_organizador) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_evento_tipo FOREIGN KEY (id_tipo_evento) REFERENCES tipo_evento(id_tipo_evento) ON DELETE RESTRICT
);

CREATE TABLE usuario_evento (
    id_usuario BIGINT NOT NULL,
    id_evento BIGINT NOT NULL,
    fecha_inscripcion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id_usuario, id_evento),
    CONSTRAINT fk_ue_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_ue_evento FOREIGN KEY (id_evento) REFERENCES evento(id_evento) ON DELETE CASCADE
);

CREATE TABLE solicitud_mentoria (
    id_mentoria BIGSERIAL PRIMARY KEY,
    id_usuario_estudiante BIGINT NOT NULL,
    id_usuario_mentor BIGINT NOT NULL,
    materia_tema VARCHAR(100) NOT NULL,
    estado_solicitud VARCHAR(30) NOT NULL DEFAULT 'PENDIENTE',
    fecha_solicitud TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_mentoria_estudiante FOREIGN KEY (id_usuario_estudiante) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    CONSTRAINT fk_mentoria_mentor FOREIGN KEY (id_usuario_mentor) REFERENCES usuario(id_usuario) ON DELETE CASCADE
);

CREATE TABLE notificacion (
    id_notificacion BIGSERIAL PRIMARY KEY,
    id_usuario BIGINT NOT NULL,
    contenido_notificacion TEXT NOT NULL,
    estado_leido BOOLEAN NOT NULL DEFAULT FALSE,
    fecha_notificacion TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_notificacion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE
);
