CREATE TABLE rol (
    id_rol INT IDENTITY(1,1) NOT NULL,
    nombre_rol NVARCHAR(30) NOT NULL,
    sigla_rol CHAR(3) NOT NULL,
    descripcion_rol NVARCHAR(150) NULL,
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_rol PRIMARY KEY (id_rol),
    CONSTRAINT uq_rol_nombre_rol UNIQUE (nombre_rol),
    CONSTRAINT uq_rol_sigla_rol UNIQUE (sigla_rol)
);


CREATE TABLE tipo_usuario (
    id_tipo_usuario INT IDENTITY(1,1) NOT NULL,
    nombre_tipo_usuario NVARCHAR(30) NOT NULL,
    sigla_tipo_usuario CHAR(3) NOT NULL,
    descripcion_tipo_usuario NVARCHAR(150) NULL,
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_tipo_usuario PRIMARY KEY (id_tipo_usuario),
    CONSTRAINT uq_tipo_usuario_nombre_tipo_usuario UNIQUE (nombre_tipo_usuario),
    CONSTRAINT uq_tipo_usuario_sigla_tipo_usuario UNIQUE (sigla_tipo_usuario)
);


CREATE TABLE usuario (
    id_usuario BIGINT IDENTITY(1,1) NOT NULL,
    uuid_usuario UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID(),
    id_rol INT NOT NULL,
    id_tipo_usuario INT NOT NULL,
    nombre_usuario VARCHAR(50) NOT NULL,
    nombres NVARCHAR(80) NOT NULL,
    apellidos NVARCHAR(80) NOT NULL,
    correo_electronico VARCHAR(120) NOT NULL,
    contrasena_hash VARCHAR(255) NOT NULL,
    fecha_nacimiento DATE NULL,
    estado_activo BIT NOT NULL DEFAULT 1,
    fecha_registro DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_usuario PRIMARY KEY (id_usuario),
    CONSTRAINT uq_usuario_uuid_usuario UNIQUE (uuid_usuario),
    CONSTRAINT uq_usuario_nombre_usuario UNIQUE (nombre_usuario),
    CONSTRAINT uq_usuario_correo_electronico UNIQUE (correo_electronico),
    CONSTRAINT fk_usuario_rol FOREIGN KEY (id_rol) REFERENCES rol (id_rol) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_usuario_tipo_usuario FOREIGN KEY (id_tipo_usuario) REFERENCES tipo_usuario (id_tipo_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE perfil (
    id_usuario BIGINT NOT NULL,
    foto_perfil VARCHAR(2048) NULL,
    biografia VARCHAR(MAX) NULL,
    programa_academico NVARCHAR(100) NULL,
    semestre_actual INT NULL,
    datos_perfil NVARCHAR(MAX) NULL,
    fecha_actualizacion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_perfil PRIMARY KEY (id_usuario),
    CONSTRAINT ck_perfil_semestre_actual CHECK (semestre_actual BETWEEN 1 AND 12),
    CONSTRAINT ck_perfil_datos_perfil CHECK (ISJSON(datos_perfil) = 1),
    CONSTRAINT fk_perfil_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE CASCADE ON UPDATE NO ACTION
);


CREATE TABLE categoria_publicacion (
    id_categoria_publicacion INT IDENTITY(1,1) NOT NULL,
    nombre_categoria_publicacion NVARCHAR(40) NOT NULL,
    descripcion_categoria_publicacion NVARCHAR(150) NULL,
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_categoria_publicacion PRIMARY KEY (id_categoria_publicacion),
    CONSTRAINT uq_categoria_publicacion_nombre_categoria_publicacion UNIQUE (nombre_categoria_publicacion)
);


CREATE TABLE publicacion (
    id_publicacion BIGINT IDENTITY(1,1) NOT NULL,
    id_usuario BIGINT NOT NULL,
    id_categoria_publicacion INT NOT NULL,
    titulo_publicacion NVARCHAR(150) NOT NULL,
    contenido_texto VARCHAR(MAX) NOT NULL,
    ruta_audio VARCHAR(255) NULL,
    fecha_publicacion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_publicacion PRIMARY KEY (id_publicacion),
    CONSTRAINT fk_publicacion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_publicacion_categoria_publicacion FOREIGN KEY (id_categoria_publicacion) REFERENCES categoria_publicacion (id_categoria_publicacion) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE comentario (
    id_publicacion BIGINT NOT NULL,
    nro_comentario INT NOT NULL,
    id_usuario BIGINT NOT NULL,
    contenido_comentario VARCHAR(MAX) NOT NULL,
    fecha_comentario DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_comentario PRIMARY KEY (id_publicacion, nro_comentario),
    CONSTRAINT fk_comentario_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion (id_publicacion) ON DELETE CASCADE ON UPDATE NO ACTION,
    CONSTRAINT fk_comentario_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE tipo_reaccion (
    id_tipo_reaccion INT IDENTITY(1,1) NOT NULL,
    nombre_tipo_reaccion NVARCHAR(40) NOT NULL,
    descripcion_tipo_reaccion NVARCHAR(150) NULL,
    estado_activo BIT NOT NULL DEFAULT 1,

    CONSTRAINT pk_tipo_reaccion PRIMARY KEY (id_tipo_reaccion),

);


CREATE TABLE reaccion (
    id_publicacion BIGINT NOT NULL,
    id_usuario BIGINT NOT NULL,
    id_tipo_reaccion INT NOT NULL,
    fecha_reaccion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_reaccion PRIMARY KEY (id_publicacion, id_usuario),
    CONSTRAINT fk_reaccion_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion (id_publicacion) ON DELETE CASCADE ON UPDATE NO ACTION,
    CONSTRAINT fk_reaccion_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE CASCADE ON UPDATE NO ACTION,

);


CREATE TABLE multimedia (
    id_publicacion BIGINT NOT NULL,
    nro_multimedia INT NOT NULL,
    tipo_archivo VARCHAR(20) NOT NULL,
    url_archivo VARCHAR(2048) NOT NULL,
    nombre_archivo NVARCHAR(150) NULL,
    fecha_carga DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_multimedia PRIMARY KEY (id_publicacion, nro_multimedia),
    CONSTRAINT ck_multimedia_tipo_archivo CHECK (tipo_archivo IN ('imagen','documento','video','audio')),
    CONSTRAINT fk_multimedia_publicacion FOREIGN KEY (id_publicacion) REFERENCES publicacion (id_publicacion) ON DELETE CASCADE ON UPDATE NO ACTION
);


CREATE TABLE tipo_grupo (
    id_tipo_grupo INT IDENTITY(1,1) NOT NULL,

CREATE TABLE grupo (
    id_grupo BIGINT IDENTITY(1,1) NOT NULL,
    id_usuario_creador BIGINT NOT NULL,
    id_tipo_grupo INT NOT NULL,
    nombre_grupo NVARCHAR(100) NOT NULL,
    descripcion_grupo VARCHAR(MAX) NULL,
    imagen_portada VARCHAR(255) NULL,
    fecha_creacion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_grupo PRIMARY KEY (id_grupo),
    CONSTRAINT uq_grupo_nombre_grupo UNIQUE (nombre_grupo),

CREATE TABLE usuario_grupo (
    id_usuario BIGINT NOT NULL,
    id_grupo BIGINT NOT NULL,
    rol_grupo VARCHAR(20) NOT NULL,
    fecha_union DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_usuario_grupo PRIMARY KEY (id_usuario, id_grupo),
    CONSTRAINT ck_usuario_grupo_rol_grupo CHECK (rol_grupo IN ('miembro','moderador','administrador')),
    CONSTRAINT fk_usuario_grupo_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE CASCADE ON UPDATE NO ACTION,
    CONSTRAINT fk_usuario_grupo_grupo FOREIGN KEY (id_grupo) REFERENCES grupo (id_grupo) ON DELETE CASCADE ON UPDATE NO ACTION
);


CREATE TABLE tipo_servicio (

CREATE TABLE servicio (
    id_servicio BIGINT IDENTITY(1,1) NOT NULL,
    id_usuario BIGINT NOT NULL,
    id_tipo_servicio INT NOT NULL,
    nombre_servicio NVARCHAR(100) NOT NULL,
    descripcion_servicio VARCHAR(MAX) NULL,
    precio_servicio DECIMAL(12,2) NOT NULL,
    disponible BIT NOT NULL DEFAULT 1,
    fecha_publicacion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_servicio PRIMARY KEY (id_servicio),
    CONSTRAINT ck_servicio_precio_servicio CHECK (precio_servicio >= 0),
    CONSTRAINT fk_servicio_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION,

CREATE TABLE tipo_producto (
    id_tipo_producto INT IDENTITY(1,1) NOT NULL,

CREATE TABLE producto (
    id_producto BIGINT IDENTITY(1,1) NOT NULL,
    id_usuario BIGINT NOT NULL,
    id_tipo_producto INT NOT NULL,
    nombre_producto NVARCHAR(100) NOT NULL,
    descripcion_producto VARCHAR(MAX) NULL,
    precio_producto DECIMAL(12,2) NOT NULL,
    stock_producto INT NOT NULL,
    imagen_producto VARCHAR(255) NULL,
    disponible BIT NOT NULL DEFAULT 1,
    fecha_publicacion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_producto PRIMARY KEY (id_producto),
    CONSTRAINT ck_producto_precio_producto CHECK (precio_producto >= 0),
    CONSTRAINT ck_producto_stock_producto CHECK (stock_producto >= 0),
    CONSTRAINT fk_producto_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_producto_tipo_producto FOREIGN KEY (id_tipo_producto) REFERENCES tipo_producto (id_tipo_producto) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE medio_pago (
    id_medio_pago INT IDENTITY(1,1) NOT NULL,
    nombre_medio_pago NVARCHAR(40) NOT NULL,
    sigla_medio_pago CHAR(3) NOT NULL,
    descripcion_medio_pago NVARCHAR(150) NULL,
    estado_activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT pk_medio_pago PRIMARY KEY (id_medio_pago),
    CONSTRAINT uq_medio_pago_nombre_medio_pago UNIQUE (nombre_medio_pago),
    CONSTRAINT uq_medio_pago_sigla_medio_pago UNIQUE (sigla_medio_pago)
);


CREATE TABLE transaccion (
    id_transaccion BIGINT IDENTITY(1,1) NOT NULL,
    uuid_transaccion UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID(),
    id_usuario_comprador BIGINT NOT NULL,
    id_medio_pago INT NOT NULL,
    monto_total DECIMAL(12,2) NOT NULL,
    estado_transaccion VARCHAR(20) NOT NULL,
    fecha_transaccion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    CONSTRAINT pk_transaccion PRIMARY KEY (id_transaccion),
    CONSTRAINT uq_transaccion_uuid_transaccion UNIQUE (uuid_transaccion),
    CONSTRAINT ck_transaccion_monto_total CHECK (monto_total >= 0),
    CONSTRAINT ck_transaccion_estado_transaccion CHECK (estado_transaccion IN ('pendiente','pagada','cancelada')),
    CONSTRAINT fk_transaccion_usuario FOREIGN KEY (id_usuario_comprador) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_transaccion_medio_pago FOREIGN KEY (id_medio_pago) REFERENCES medio_pago (id_medio_pago) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE detalle_transaccion (
    id_transaccion BIGINT NOT NULL,
    nro_detalle INT NOT NULL,
    id_producto BIGINT NULL,
    id_servicio BIGINT NULL,
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(12,2) NOT NULL,
    CONSTRAINT pk_detalle_transaccion PRIMARY KEY (id_transaccion, nro_detalle),
    CONSTRAINT ck_detalle_transaccion_cantidad CHECK (cantidad > 0),
    CONSTRAINT ck_detalle_transaccion_precio_unitario CHECK (precio_unitario >= 0),
    CONSTRAINT ck_detalle_producto_o_servicio CHECK ((id_producto IS NOT NULL AND id_servicio IS NULL) OR (id_producto IS NULL AND id_servicio IS NOT NULL)),
    CONSTRAINT fk_detalle_transaccion_transaccion FOREIGN KEY (id_transaccion) REFERENCES transaccion (id_transaccion) ON DELETE CASCADE ON UPDATE NO ACTION,
    CONSTRAINT fk_detalle_transaccion_producto FOREIGN KEY (id_producto) REFERENCES producto (id_producto) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_detalle_transaccion_servicio FOREIGN KEY (id_servicio) REFERENCES servicio (id_servicio) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE tipo_evento (

    nombre_tipo_evento NVARCHAR(40) NOT NULL,
    descripcion_tipo_evento NVARCHAR(150) NULL,

    CONSTRAINT uq_tipo_evento_nombre_tipo_evento UNIQUE (nombre_tipo_evento)

CREATE TABLE evento (
    id_evento BIGINT IDENTITY(1,1) NOT NULL,
    id_usuario_organizador BIGINT NOT NULL,
    id_tipo_evento INT NOT NULL,
    nombre_evento NVARCHAR(100) NOT NULL,
    descripcion_evento VARCHAR(MAX) NULL,
    fecha_evento DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    hora_fin TIME NULL,
    lugar_evento NVARCHAR(150) NULL,
    CONSTRAINT pk_evento PRIMARY KEY (id_evento),
    CONSTRAINT fk_evento_usuario FOREIGN KEY (id_usuario_organizador) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_evento_tipo_evento FOREIGN KEY (id_tipo_evento) REFERENCES tipo_evento (id_tipo_evento) ON DELETE NO ACTION ON UPDATE NO ACTION
);


CREATE TABLE usuario_evento (
    id_usuario BIGINT NOT NULL,
    id_evento BIGINT NOT NULL,
    fecha_inscripcion DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    asistio BIT NOT NULL DEFAULT 0,
    CONSTRAINT pk_usuario_evento PRIMARY KEY (id_usuario, id_evento),
    CONSTRAINT fk_usuario_evento_usuario FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario) ON DELETE CASCADE ON UPDATE NO ACTION,
    CONSTRAINT fk_usuario_evento_evento FOREIGN KEY (id_evento) REFERENCES evento (id_evento) ON DELETE CASCADE ON UPDATE NO ACTION
);


CREATE TABLE solicitud_mentoria (
    id_solicitud_mentoria BIGINT IDENTITY(1,1) NOT NULL,
    id_usuario_mentor BIGINT NOT NULL,
    id_usuario_estudiante BIGINT NOT NULL,
    tema_mentoria NVARCHAR(150) NOT NULL,
    mensaje_solicitud VARCHAR(MAX) NULL,
    estado_solicitud VARCHAR(20) NOT NULL,
    fecha_solicitud DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
    fecha_respuesta DATETIMEOFFSET NULL,
    CONSTRAINT pk_solicitud_mentoria PRIMARY KEY (id_solicitud_mentoria),
    CONSTRAINT ck_solicitud_mentoria_estado_solicitud CHECK (estado_solicitud IN ('pendiente','aceptada','rechazada')),
    CONSTRAINT ck_solicitud_mentoria_mentor_distinto CHECK (id_usuario_mentor <> id_usuario_estudiante),
    CONSTRAINT fk_solicitud_mentoria_usuario_id_usuario_mentor FOREIGN KEY (id_usuario_mentor) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION,
    CONSTRAINT fk_solicitud_mentoria_usuario_id_usuario_estudiante FOREIGN KEY (id_usuario_estudiante) REFERENCES usuario (id_usuario) ON DELETE NO ACTION ON UPDATE NO ACTION
);
